cmake_minimum_required(VERSION 3.16)
project(MultiCameraCompatibility LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core calib3d imgcodecs imgproc)
find_package(Threads REQUIRED)
add_executable(compatibility_test compatibility_test.cpp)
target_compile_features(compatibility_test PRIVATE cxx_std_17)
target_include_directories(compatibility_test PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(compatibility_test PRIVATE ${OpenCV_LIBS})
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/calibration_service.cpp")
    add_executable(calibration_service calibration_service.cpp)
    target_compile_features(calibration_service PRIVATE cxx_std_17)
    target_include_directories(calibration_service PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(calibration_service PRIVATE ${OpenCV_LIBS} Threads::Threads)
endif()
